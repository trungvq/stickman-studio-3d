import { it } from "vitest";
import { GRIP_GROUPS } from "./graspGroups";
import { graspFrame, type GraspChoice } from "./graspSolve";
import { HAND_PIECES, handFrames } from "./hand";
import { handSkin, skinHand } from "./handMesh";
import { heldOppose } from "./heldAudit";
import { capsuleClearance, heldFit, skinBulge, skinDepth, skinPieces, toItem } from "./heldFit";
import { handDims, itemGeom, type Feature } from "./heldGeom";
import { parseHeldItem, type GroupId, type HeldItemSpec } from "./heldItem";
import { matVec, type V3 } from "./lens";
import chaiDia from "./__fixtures__/held-v2/chai-dia.json";
import ongNghe from "./__fixtures__/held-v2/ong-nghe-ban.json";

type Pick = GraspChoice & { group: GroupId };
const item = (json: unknown): HeldItemSpec => {
  const r = parseHeldItem(json);
  if (!r.success) throw new Error(r.error.message);
  return r.data;
};
const rod = (r: number) =>
  item({
    schemaVersion: 2,
    size: [2 * r, 40, 2 * r],
    front: [0, 0, 1],
    top: [0, 1, 0],
    anchors: {},
    shape: [{ cyl: r, from: [0, 0, 0], to: [0, 40, 0] }],
  });
const HANDS = [
  [1.75, 1.68],
  [2, 1.68],
] as const;

export function measure(spec: HeldItemSpec, pick: Pick, side: number, k: number, long: number) {
  const geom = itemGeom(spec);
  const g = graspFrame(spec, geom, pick, side, handDims(k, long), k, long);
  const skin = handSkin(side, k, long);
  const fit = heldFit({
    v: g.pose,
    side,
    k,
    long,
    shape: spec.shape!,
    G: g.G,
    roles: g.roles,
    bulge: skinBulge(skin, g.pose),
  });
  const v = [...g.pose, ...fit];
  const pos = new Float32Array(skin.pos.length);
  skinHand(skin, v, pos);
  const d = skinDepth(pos, skinPieces(skin), g.G, spec.shape!);
  const F = handFrames(v, side, k, long);
  const end = (n: (typeof HAND_PIECES)[number]): V3 => {
    const e = matVec(F[n].R, [0, -F[n].len, 0]);
    return [F[n].o[0] + e[0], F[n].o[1] + e[1], F[n].o[2] + e[2]];
  };
  const pierce = Math.max(
    ...HAND_PIECES.filter((n) => n !== "palm").map(
      (n) => capsuleClearance(spec.shape!, g.G, F[n].o, end(n), 0).core,
    ),
  );
  const f = geom.features[g.choice.featureIndex!] as Feature & { a: V3; c?: V3 };
  const a = f.a;
  const tipT = toItem(g.G, end("thumb1"));
  const baseT = toItem(g.G, F.thumb0.o);
  const tipI = toItem(g.G, end("index1"));
  const palmC = toItem(g.G, [0, -0.05, 0]);
  const dt = tipT.map((x, i) => x - baseT[i]);
  const thumbUp =
    (dt[0] * a[0] + dt[1] * a[1] + dt[2] * a[2]) / Math.hypot(dt[0], dt[1], dt[2]);
  // góc quanh trục (độ) của đầu ngón cái / trỏ so với tâm lòng tay, trục đi qua tâm khối hình (chiếu)
  const ang = (q: V3) => {
    const c: V3 = (f as unknown as { c: V3 }).c ?? [0, 0, 0];
    const proj = (p: V3) => {
      const w = p.map((x, i) => x - c[i]);
      const s = w[0] * a[0] + w[1] * a[1] + w[2] * a[2];
      return w.map((x, i) => x - s * a[i]) as V3;
    };
    const u = proj(palmC);
    const w = proj(q);
    const cr = [
      u[1] * w[2] - u[2] * w[1],
      u[2] * w[0] - u[0] * w[2],
      u[0] * w[1] - u[1] * w[0],
    ];
    const s = Math.sign(cr[0] * a[0] + cr[1] * a[1] + cr[2] * a[2]) || 1;
    const cos = (u[0] * w[0] + u[1] * w[1] + u[2] * w[2]) / Math.hypot(...u) / Math.hypot(...w);
    return (s * Math.acos(Math.max(-1, Math.min(1, cos)))) / (Math.PI / 180);
  };
  return {
    choice: g.choice,
    sink: Math.max(0, -Math.min(...Object.values(d.gaps))),
    pierce,
    gaps: d.gaps,
    oppose: heldOppose(spec.shape!, d.near, d.gaps, g.roles),
    thumbUp,
    thumbAng: ang(tipT),
    indexAng: ang(tipI),
    pose: g.pose.slice(2, 6).map((x) => Math.round(x)),
  };
}

const r1 = (x: number) => Math.round(x * 10) / 10;
const line = (tag: string, m: ReturnType<typeof measure>) =>
  ((...a: unknown[]) => process.stderr.write(a.join(" ") + "\n"))(
    tag,
    "sink",
    r1(m.sink),
    "pierce",
    r1(m.pierce),
    "thumbGap",
    r1(m.gaps.thumb),
    "palm",
    r1(m.gaps.palm),
    "idx",
    r1(m.gaps.index),
    "pinky",
    r1(m.gaps.pinky),
    "opp",
    m.oppose ? `${Math.round(m.oppose.angle)}/${m.oppose.part}` : "-",
    "up",
    r1(m.thumbUp),
    "angT",
    Math.round(m.thumbAng),
    "angI",
    Math.round(m.indexAng),
    "pose",
    m.pose.join(","),
  );

it("sweep", () => {
  const env = process.env.SWEEP ?? "base";
  const tab = process.env.TAB;
  if (tab) (GRIP_GROUPS.cylinder.variants.wrap.pose as { thumb: unknown }).thumb = JSON.parse(tab);
  const CHAI = item(chaiDia);
  const ONG = item(ongNghe);
  for (const [k, long] of HANDS)
    for (const side of [-1, 1]) {
      const tag = `${env} k${k} s${side}`;
      for (const r of (process.env.RR ?? '2.5,5,8,10,13').split(',').map(Number))
        line(`${tag} rod${r}`, measure(rod(r), { group: "cylinder", around: 0 }, side, k, long));
      line(
        `${tag} chai than`,
        measure(CHAI, { group: "cylinder", part: "than", along: 0.25, around: 300 }, side, k, long),
      );
      line(
        `${tag} chai vai`,
        measure(CHAI, { group: "cylinder", part: "vai", along: 0.93, around: 180 }, side, k, long),
      );
      for (const around of [150, 180])
        line(
          `${tag} ong flat ${around}`,
          measure(ONG, { group: "cylinder", variant: "flat", around }, side, k, long),
        );
      for (const around of [90, 180])
        line(
          `${tag} ong wrap ${around}`,
          measure(ONG, { group: "cylinder", variant: "wrap", around }, side, k, long),
        );
    }
});

it("dir", () => {
  if (!process.env.DIR) return;
  for (const abd of [0, 30, 60, 90])
    for (const opp of [0, 45, 90, 135])
      for (const tb of [0, 45]) {
        const v = new Array(20).fill(0);
        v[2] = abd;
        v[3] = opp;
        v[4] = tb;
        const F = handFrames(v, -1, 1, 1);
        const d = matVec(F.thumb0.R, [0, -1, 0]);
        process.stderr.write(
          `D ${abd},${opp},${tb} -> ${d.map((x) => x.toFixed(2)).join(",")}\n`,
        );
      }
});

it("grid", () => {
  if (!process.env.GRID) return;
  const out: string[] = [];
  const P = GRIP_GROUPS.cylinder.variants.wrap.pose as { thumb: unknown };
  const rs = (process.env.RS ?? "8,10,13").split(",").map(Number);
  const A = (process.env.ABD ?? "20,40,60,80,100").split(",").map(Number);
  const O = (process.env.OPP ?? "30,50,70,90,110").split(",").map(Number);
  const B = (process.env.TB ?? "-20,0,20,40").split(",").map(Number);
  const T = (process.env.TT ?? "0,20,40").split(",").map(Number);
  for (const abd of A)
    for (const opp of O)
      for (const tb of B)
        for (const tt of T) {
          P.thumb = [[0.5, [abd, opp, tb, tt]]];
          let worst = { opp: 999, ang: -999, up: 0, sink: 0, pierce: -9, gap: -9 };
          for (const r of rs)
            for (const [k, long] of HANDS) {
              const m = measure(rod(r), { group: "cylinder", around: 0 }, -1, k, long);
              worst = {
                opp: Math.min(worst.opp, m.oppose?.angle ?? 0),
                ang: Math.max(worst.ang, m.thumbAng),
                up: Math.max(worst.up, Math.abs(m.thumbUp)),
                sink: Math.max(worst.sink, m.sink),
                pierce: Math.max(worst.pierce, m.pierce),
                gap: Math.max(worst.gap, m.gaps.thumb),
              };
            }
          out.push(
            [abd, opp, tb, tt].join(",") +
              ` opp ${Math.round(worst.opp)} ang ${Math.round(worst.ang)} up ${r1(worst.up)} sink ${r1(worst.sink)} pierce ${r1(worst.pierce)} gap ${r1(worst.gap)}`,
          );
        }
  process.stderr.write(out.map((l) => "G " + l).join("\n") + "\n");
});
